/* ─── Helper: print rwxrwxrwx style permissions ─── */
void print_permissions(mode_t mode) {
    printf((S_ISDIR(mode))  ? "d" : "-");
    printf((mode & S_IRUSR) ? "r" : "-");
    printf((mode & S_IWUSR) ? "w" : "-");
    printf((mode & S_IXUSR) ? "x" : "-");
    printf((mode & S_IRGRP) ? "r" : "-");
    printf((mode & S_IWGRP) ? "w" : "-");
    printf((mode & S_IXGRP) ? "x" : "-");
    printf((mode & S_IROTH) ? "r" : "-");
    printf((mode & S_IWOTH) ? "w" : "-");
    printf((mode & S_IXOTH) ? "x" : "-");
}

/* ─── List all entries in a directory ─── */
void list_directory(const char *path) {
    DIR *dir = opendir(path);          /* system call: opendir */
    if (!dir) {
        perror("opendir failed");
        return;
    }

    struct dirent *entry;
    struct stat file_stat;
    char full_path[1024];

    printf("\n%-30s %-12s %-12s %s\n", "Name", "Type", "Size", "Permissions");
    printf("%-30s %-12s %-12s %s\n",
           "------------------------------",
           "------------",
           "------------",
           "----------");

    while ((entry = readdir(dir)) != NULL) {   /* system call: readdir */
        /* skip hidden . and .. */
        if (strcmp(entry->d_name, ".") == 0 || strcmp(entry->d_name, "..") == 0)
            continue;

        snprintf(full_path, sizeof(full_path), "%s/%s", path, entry->d_name);
        stat(full_path, &file_stat);           /* system call: stat */

        /* Type */
        char *type = S_ISDIR(file_stat.st_mode) ? "Directory" : "File";

        /* Size in human-readable form */
        char size_buf[20];
        if (file_stat.st_size >= 1024 * 1024)
            snprintf(size_buf, sizeof(size_buf), "%.1f MB",
                     file_stat.st_size / (1024.0 * 1024.0));
        else if (file_stat.st_size >= 1024)
            snprintf(size_buf, sizeof(size_buf), "%.1f KB",
                     file_stat.st_size / 1024.0);
        else
            snprintf(size_buf, sizeof(size_buf), "%ld B", file_stat.st_size);

        printf("%-30s %-12s %-12s ", entry->d_name, type, size_buf);
        print_permissions(file_stat.st_mode);
        printf("\n");
    }

    closedir(dir);    /* system call: closedir */
    printf("\n");
}

/* ─── Change current working directory ─── */
void change_directory(char *current_path) {
    char new_dir[512];
    printf("Enter directory name (or '..' to go up): ");
    scanf("%s", new_dir);

    char new_path[1024];
    if (strcmp(new_dir, "..") == 0) {
        /* go up one level */
        char *last_slash = strrchr(current_path, '/');
        if (last_slash && last_slash != current_path)
            *last_slash = '\0';
        else
            strcpy(current_path, "/");
    } else {
        snprintf(new_path, sizeof(new_path), "%s/%s", current_path, new_dir);
        struct stat st;
        if (stat(new_path, &st) == 0 && S_ISDIR(st.st_mode)) {
            strcpy(current_path, new_path);
        } else {
            printf("[!] '%s' is not a valid directory.\n", new_dir);
        }
    }
}

/* ─── Show detailed properties of a single file ─── */
void view_file_properties(const char *current_path) {
    char filename[512];
    printf("Enter file/folder name: ");
    scanf("%s", filename);

    char full_path[1024];
    snprintf(full_path, sizeof(full_path), "%s/%s", current_path, filename);

    struct stat st;
    if (stat(full_path, &st) != 0) {     /* system call: stat */
        perror("stat failed");
        return;
    }

    printf("\n======= File Properties =======\n");
    printf("  Name       : %s\n", filename);
    printf("  Type       : %s\n", S_ISDIR(st.st_mode) ? "Directory" : "Regular File");
    printf("  Size       : %ld bytes\n", st.st_size);
    printf("  Permissions: ");
    print_permissions(st.st_mode);
    printf("\n");

    /* Owner */
    struct passwd *pw = getpwuid(st.st_uid);
    struct group  *gr = getgrgid(st.st_gid);
    printf("  Owner      : %s\n", pw ? pw->pw_name : "unknown");
    printf("  Group      : %s\n", gr ? gr->gr_name : "unknown");

    /* Timestamps */
    char timebuf[64];
    strftime(timebuf, sizeof(timebuf), "%Y-%m-%d %H:%M:%S",
             localtime(&st.st_mtime));
    printf("  Modified   : %s\n", timebuf);

    strftime(timebuf, sizeof(timebuf), "%Y-%m-%d %H:%M:%S",
             localtime(&st.st_atime));
    printf("  Accessed   : %s\n", timebuf);
    printf("================================\n\n");
}
