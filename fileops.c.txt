#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include "fileops.h"

#define BUFFER_SIZE 4096

/* ─── Copy a file using read/write system calls ─── */
void copy_file(const char *current_path) {
    char src_name[512], dst_name[512];

    printf("Enter source filename: ");
    scanf("%s", src_name);
    printf("Enter destination filename: ");
    scanf("%s", dst_name);

    char src_path[1024], dst_path[1024];
    snprintf(src_path, sizeof(src_path), "%s/%s", current_path, src_name);
    snprintf(dst_path, sizeof(dst_path), "%s/%s", current_path, dst_name);

    /* Open source for reading */
    int src_fd = open(src_path, O_RDONLY);          /* system call: open */
    if (src_fd < 0) {
        perror("Cannot open source file");
        return;
    }

    /* Open destination for writing (create if not exists) */
    int dst_fd = open(dst_path, O_WRONLY | O_CREAT | O_TRUNC, 0644); /* system call: open */
    if (dst_fd < 0) {
        perror("Cannot create destination file");
        close(src_fd);
        return;
    }

    /* Read from source and write to destination in chunks */
    char buffer[BUFFER_SIZE];
    ssize_t bytes_read;
    long total = 0;

    while ((bytes_read = read(src_fd, buffer, BUFFER_SIZE)) > 0) { /* system call: read */
        ssize_t bytes_written = write(dst_fd, buffer, bytes_read); /* system call: write */
        if (bytes_written != bytes_read) {
            perror("Write error");
            break;
        }
        total += bytes_written;
    }

    close(src_fd);   /* system call: close */
    close(dst_fd);   /* system call: close */

    printf("[✓] File copied successfully! (%ld bytes written)\n\n", total);
}

/* ─── Delete a file using unlink system call ─── */
void delete_file(const char *current_path) {
    char filename[512];
    printf("Enter filename to delete: ");
    scanf("%s", filename);

    char full_path[1024];
    snprintf(full_path, sizeof(full_path), "%s/%s", current_path, filename);

    /* Confirm before deleting */
    char confirm;
    printf("[!] Are you sure you want to delete '%s'? (y/n): ", filename);
    scanf(" %c", &confirm);

    if (confirm == 'y' || confirm == 'Y') {
        if (remove(full_path) == 0) {        /* system call: remove/unlink */
            printf("[✓] File '%s' deleted successfully.\n\n", filename);
        } else {
            perror("Delete failed");
        }
    } else {
        printf("[~] Delete cancelled.\n\n");
    }
}

/* ─── Rename a file using rename system call ─── */
void rename_file(const char *current_path) {
    char old_name[512], new_name[512];

    printf("Enter current filename: ");
    scanf("%s", old_name);
    printf("Enter new filename: ");
    scanf("%s", new_name);

    char old_path[1024], new_path[1024];
    snprintf(old_path, sizeof(old_path), "%s/%s", current_path, old_name);
    snprintf(new_path, sizeof(new_path), "%s/%s", current_path, new_name);

    if (rename(old_path, new_path) == 0) {   /* system call: rename */
        printf("[✓] Renamed '%s' → '%s' successfully.\n\n", old_name, new_name);
    } else {
        perror("Rename failed");
    }
}
